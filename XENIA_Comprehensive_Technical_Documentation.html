2.2 Request Flow Architecture

Standard Request Flow:

  1. Client Request → Browser/App initiates HTTP/WebSocket request
  2. CDN Layer → Static assets served from edge locations
  3. Load Balancer → Distributes traffic across server instances
  4. API Gateway → Authentication, rate limiting, routing
  5. Application Server → Business logic processing
  6. Service Layer → Specialized service handling
  7. Data Layer → Database operations
  8. Response Pipeline → Reverse flow with caching

Real-time Update Flow:

  1. WebSocket Connection → Persistent connection established
  2. Supabase Realtime → PostgreSQL changes trigger events
  3. Event Broadcasting → Updates pushed to subscribed clients
  4. UI Update → React components re-render with new data

    limit_count INT DEFAULT 10

RETURNS TABLE(topic VARCHAR, similarity FLOAT)
AS $$
BEGIN
    RETURN QUERY
    SELECT 
        te.topic,
        1 - (te.embedding <=> query_embedding) as similarity
    FROM topic_embeddings te
    ORDER BY te.embedding <=> query_embedding
    LIMIT limit_count;
END;

6.3 Circuit Breaker Pattern

Implementation Details

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=60):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED

    def call(self, func, *args, **kwargs):
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
            else:
                raise CircuitOpenException()

        try:
            result = func(*args, **kwargs)
            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.CLOSED
                self.failure_count = 0
            return result
        except Exception as e:
            self.failure_count += 1
            self.last_failure_time = time.time()

            if self.failure_count >= self.failure_threshold:
                self.state = CircuitState.OPEN

            raise e

// Token Management
class AuthService {
  private readonly SECRET_KEY = process.env.JWT_SECRET;
  private readonly REFRESH_SECRET = process.env.JWT_REFRESH_SECRET;

  generateTokens(userId: string, role: string) {
    const accessToken = jwt.sign(
      { userId, role, type: 'access' },
      this.SECRET_KEY,
      { expiresIn: '15m' }
    );

    const refreshToken = jwt.sign(
      { userId, type: 'refresh' },
      this.REFRESH_SECRET,
      { expiresIn: '7d' }
    );

    return { accessToken, refreshToken };
  }

  async validateToken(token: string): Promise<TokenPayload> {
    try {
      const payload = jwt.verify(token, this.SECRET_KEY);

      // Additional validation
      const user = await this.getUserFromDatabase(payload.userId);
      if (!user || user.status !== 'active') {
        throw new UnauthorizedException();
      }

      return payload;
    } catch (error) {
      throw new InvalidTokenException();
    }
  }
}

7.3 Rate Limiting

Multi-level Rate Limiting

class RateLimiter:
    def __init__(self):
        self.limits = {
            'api': {'requests': 100, 'window': 60},      # 100/min
            'ai': {'requests': 20, 'window': 60},        # 20/min
            'upload': {'requests': 10, 'window': 3600},  # 10/hour
            'auth': {'requests': 5, 'window': 300}       # 5/5min
        }

    def check_limit(self, user_id: str, endpoint_type: str) -> bool:
        limit = self.limits.get(endpoint_type, self.limits['api'])
        key = f"rate:{endpoint_type}:{user_id}:{int(time.time() // limit['window'])}"

        count = redis_client.incr(key)
        if count == 1:
            redis_client.expire(key, limit['window'])

        if count > limit['requests']:
            raise RateLimitExceededException(
                f"Rate limit exceeded: {limit['requests']}/{limit['window']}s"
            )

        return True

Database Query Optimization

# N+1 Query Prevention
class OptimizedQueries:
    @staticmethod
    def get_user_with_plans(user_id: str):
        # Bad: N+1 queries
        # user = db.query(User).filter_by(id=user_id).first()
        # for plan in user.plans:
        #     tasks = db.query(Task).filter_by(plan_id=plan.id).all()

        # Good: Single query with joins
        query = """
        SELECT 
            u.*,
            array_agg(
                json_build_object(
                    'id', p.id,
                    'title', p.title,
                    'tasks', p.tasks
                )
            ) as plans
        FROM users u
        LEFT JOIN study_plans p ON u.id = p.user_id
        LEFT JOIN LATERAL (
            SELECT array_agg(t.*) as tasks
            FROM tasks t
            WHERE t.plan_id = p.id
        ) t ON true
        WHERE u.id = %s
        GROUP BY u.id
        """
        return db.execute(query, [user_id]).fetchone()

Caching Strategy

class CacheStrategy:
    # Multi-level caching
    def get_with_cache(self, key: str, compute_func):
        # L1: Local memory cache (instant)
        if key in self.local_cache:
            return self.local_cache[key]

        # L2: Redis cache (1-5ms)
        redis_value = redis_client.get(key)
        if redis_value:
            value = json.loads(redis_value)
            self.local_cache[key] = value
            return value

        # L3: Database (10-50ms)
        value = compute_func()

        # Cache in both layers
        self.local_cache[key] = value
        redis_client.setex(key, 300, json.dumps(value))

        return value

10. TECHNICAL DECISION RATIONALE

10.1 Architecture Decisions

Monolith vs Microservices

Decision: Modular Monolith - Rationale: Start simple, scale when needed - Benefits: Easier deployment, debugging, and maintenance - Future Path: Service extraction when traffic exceeds 100k users

Sync vs Async Processing

Decision: Hybrid Approach - Synchronous: User-facing APIs (<2s response) - Asynchronous: AI processing, file uploads, batch operations - Implementation: Celery for async tasks

10.2 Technology Choices

Why Not Node.js Backend?

Python Chosen Because: - AI Ecosystem: Better library support - Data Processing: Superior text manipulation - Team Expertise: Existing Python knowledge - Performance: Adequate for current scale

Why Not GraphQL?

REST Chosen Because: - Simplicity: Easier to understand and debug - Caching: Better HTTP caching support - Tools: More mature ecosystem - Team Experience: Familiar with REST

Why Not MongoDB?

PostgreSQL Chosen Because: - ACID Compliance: Transaction safety critical - Relationships: Complex data relationships - SQL: Powerful query capabilities - Supabase: Real-time features included
